cmake_minimum_required(VERSION 3.25)
project(mercenaries2_gpu_patch LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.10.0 REQUIRED CONFIG)
set(REXGLUE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../reference/rexglue-upstream")
set(REXGLUE_USE_D3D12 ON)
set(REXGLUE_USE_VULKAN OFF)
add_library(rexruntime ALIAS rex::runtime)
add_library(xxHash::xxhash STATIC IMPORTED)
set_target_properties(xxHash::xxhash PROPERTIES IMPORTED_LOCATION "${REX_PORTABLE_SDK}/lib/xxhash.lib")
function(rex_target_pch target)
  target_precompile_headers(${target} PRIVATE ${ARGN})
endfunction()
add_compile_definitions(REX_PLATFORM_WINDOWS=1)
add_compile_options(-msse4.1)
add_subdirectory("${REXGLUE_ROOT}/src/graphics" graphics)
